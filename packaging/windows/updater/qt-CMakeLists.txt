cmake_minimum_required(VERSION 3.21)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Widgets Svg)
if(UNIX AND NOT APPLE)
    find_package(Qt6 REQUIRED COMPONENTS DBus)
endif()
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CEDAR_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/..")
list(APPEND CMAKE_MODULE_PATH "${CEDAR_SOURCE}/cmake/module")
include(SetupSkia)
find_package(OpenGL REQUIRED)
add_subdirectory("${CEDAR_SOURCE}/logic" "${CMAKE_CURRENT_BINARY_DIR}/logic")
add_subdirectory("${CEDAR_SOURCE}/format" "${CMAKE_CURRENT_BINARY_DIR}/format")
add_library(CedarModel STATIC
    ${CEDAR_SOURCE}/src/gui/guiGate.cpp
    ${CEDAR_SOURCE}/src/gui/guiWire.cpp
    ${CEDAR_SOURCE}/src/gui/wire/SegmentMap.cpp
    ${CEDAR_SOURCE}/src/gui/wire/WireTopology.cpp
    ${CEDAR_SOURCE}/src/gui/wire/SegmentDrag.cpp
    ${CEDAR_SOURCE}/src/gui/route/TrunkRouter.cpp
    ${CEDAR_SOURCE}/src/gui/wireSegment.cpp
    ${CEDAR_SOURCE}/src/gui/klsCollisionChecker.cpp
    ${CEDAR_SOURCE}/src/gui/gl_defs.cpp
    ${CEDAR_SOURCE}/src/gui/XMLParser.cpp
    ${CEDAR_SOURCE}/src/gui/LibraryParse.cpp
    ${CEDAR_SOURCE}/src/gui/render/SkiaBackend.cpp
    ${CEDAR_SOURCE}/src/gui/render/SkiaScene.cpp
    SharedModel.cpp
    ${CEDAR_SOURCE}/src/gui/command/cmdSerialize.cpp)
target_include_directories(CedarModel PUBLIC ${CEDAR_SOURCE}/include/gui PRIVATE ${CEDAR_SOURCE}/include/gui/command)
target_link_libraries(CedarModel PUBLIC Logic CircuitFile cedar_skia OpenGL::GL)
qt_add_executable(CedarLogic main.cpp CrashReporter.cpp CrashReporter.h Keyboard.cpp Keyboard.h DocumentLock.cpp Appearance.cpp Appearance.h Editor.cpp Editor_Desktop.cpp Editor.h Canvas.cpp Canvas.h Simulation.cpp Simulation.h)
qt_add_resources(CedarLogic "frontend" PREFIX "/" FILES qml/Main.qml qml/PartTile.qml qml/Tool.qml qml/Preferences.qml qml/AppCombo.qml qml/AppField.qml qml/AppButton.qml qml/QuickAdd.qml qml/Tutorial.qml qml/ShortcutHelp.qml)
target_link_libraries(CedarLogic PRIVATE CedarModel Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Svg)
if(UNIX AND NOT APPLE)
    target_link_libraries(CedarLogic PRIVATE Qt6::DBus)
    target_compile_definitions(CedarLogic PRIVATE CEDAR_PORTAL_THEME)
endif()
target_compile_definitions(CedarLogic PRIVATE CEDAR_SOURCE_RES="${CEDAR_SOURCE}/res")
target_compile_definitions(CedarLogic PRIVATE CEDAR_RELEASE_VERSION="3.08")
add_custom_command(TARGET CedarLogic POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEDAR_SOURCE}/res" "$<TARGET_FILE_DIR:CedarLogic>/res")
if(WIN32)
    set(WINSPARKLE_ROOT "${CEDAR_SOURCE}/../cache/winsparkle/WinSparkle-0.9.4" CACHE PATH "WinSparkle 0.9.4 SDK")
    find_path(WINSPARKLE_INCLUDE_DIR winsparkle.h HINTS "${WINSPARKLE_ROOT}/include" REQUIRED)
    find_library(WINSPARKLE_LIBRARY NAMES WinSparkle HINTS "${WINSPARKLE_ROOT}/x64/Release" REQUIRED)
    target_sources(CedarLogic PRIVATE WindowsUpdater.cpp WindowsUpdater.h WindowsUpdateKey.h)
    target_include_directories(CedarLogic PRIVATE "${WINSPARKLE_INCLUDE_DIR}")
    target_link_libraries(CedarLogic PRIVATE "${WINSPARKLE_LIBRARY}")
    set(CEDAR_WINDOWS_BUILD_REVISION 1)
    set(CEDAR_WINDOWS_BUILD_VERSION "${PROJECT_VERSION}.${CEDAR_WINDOWS_BUILD_REVISION}")
    target_compile_definitions(CedarLogic PRIVATE CEDAR_WINDOWS_BUILD_VERSION="${CEDAR_WINDOWS_BUILD_VERSION}")
    add_custom_command(TARGET CedarLogic POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINSPARKLE_ROOT}/x64/Release/WinSparkle.dll" "$<TARGET_FILE_DIR:CedarLogic>/WinSparkle.dll")
    if(BUILD_TESTING)
        qt_add_executable(test_qt_updater updater_tests.cpp WindowsUpdater.cpp WindowsUpdater.h WindowsUpdateKey.h)
        target_include_directories(test_qt_updater PRIVATE "${WINSPARKLE_INCLUDE_DIR}")
        target_link_libraries(test_qt_updater PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets "${WINSPARKLE_LIBRARY}")
        target_compile_definitions(test_qt_updater PRIVATE CEDAR_RELEASE_VERSION="3.08" CEDAR_WINDOWS_BUILD_VERSION="${CEDAR_WINDOWS_BUILD_VERSION}")
        add_executable(test_update_installer_probe update_installer_probe.cpp)
    endif()
    add_custom_command(TARGET CedarLogic POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy "${CEDAR_SOURCE}/res/windows/LabelFont.ttf" "$<TARGET_FILE_DIR:CedarLogic>/res/LabelFont.ttf"
        COMMAND ${CMAKE_COMMAND} -E copy "${CEDAR_SOURCE}/res/windows/LabelFont-license.txt" "$<TARGET_FILE_DIR:CedarLogic>/res/LabelFont-license.txt")
    enable_language(RC)
    configure_file("${CEDAR_SOURCE}/res/version.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
    target_sources(CedarLogic PRIVATE ${CEDAR_SOURCE}/icon.rc "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
    target_include_directories(CedarLogic PRIVATE ${CEDAR_SOURCE})
    set_target_properties(CedarLogic PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
install(TARGETS CedarLogic RUNTIME DESTINATION bin)
install(DIRECTORY "${CEDAR_SOURCE}/res/" DESTINATION share/CedarLogic/res)
if(BUILD_TESTING)
    add_subdirectory("${CEDAR_SOURCE}/src/gui/tests" "${CMAKE_CURRENT_BINARY_DIR}/model-tests")
    add_subdirectory("${CEDAR_SOURCE}/src/gui/route/tests" "${CMAKE_CURRENT_BINARY_DIR}/route-tests")
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(test_qt_ui ui_tests.cpp Keyboard.cpp Keyboard.h DocumentLock.cpp Editor.cpp Editor_Desktop.cpp Editor.h Appearance.cpp Appearance.h Canvas.cpp Canvas.h Simulation.cpp Simulation.h)
    qt_add_resources(test_qt_ui "ui_frontend" PREFIX "/" FILES qml/Main.qml qml/PartTile.qml qml/Tool.qml qml/Preferences.qml qml/AppCombo.qml qml/AppField.qml qml/AppButton.qml qml/QuickAdd.qml qml/Tutorial.qml qml/ShortcutHelp.qml)
    target_link_libraries(test_qt_ui PRIVATE CedarModel Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Svg Qt6::Test)
    target_compile_definitions(test_qt_ui PRIVATE CEDAR_SOURCE_RES="${CEDAR_SOURCE}/res")
    if(UNIX AND NOT APPLE)
        target_link_libraries(test_qt_ui PRIVATE Qt6::DBus)
        target_compile_definitions(test_qt_ui PRIVATE CEDAR_PORTAL_THEME)
    endif()
    if(UNIX AND NOT APPLE)
        add_executable(test_qt_portal portal_tests.cpp)
        target_link_libraries(test_qt_portal PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus)
        add_test(NAME qt_native_file_picker COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/test_portal.sh" "$<TARGET_FILE:test_qt_portal>" "${CEDAR_SOURCE}/format/tests/fixtures/lab6.cdl" "${CMAKE_CURRENT_BINARY_DIR}/portal-test.log")
        set_tests_properties(qt_native_file_picker PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    endif()
    add_test(NAME qt_keyboard_and_layout COMMAND test_qt_ui)
    set_tests_properties(qt_keyboard_and_layout PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    add_test(NAME qt_ui_smoke COMMAND CedarLogic --theme dark --screenshot "${CMAKE_CURRENT_BINARY_DIR}/ui-smoke.png")
    set_tests_properties(qt_ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config")
    add_executable(test_qt_editor editor_tests.cpp DocumentLock.cpp Editor.cpp Editor_Desktop.cpp Editor.h Appearance.cpp Appearance.h Simulation.cpp Simulation.h)
    target_link_libraries(test_qt_editor PRIVATE CedarModel Qt6::Core Qt6::Gui Qt6::Widgets)
    target_compile_definitions(test_qt_editor PRIVATE CEDAR_SOURCE_RES="${CEDAR_SOURCE}/res")
    if(UNIX AND NOT APPLE)
        target_link_libraries(test_qt_editor PRIVATE Qt6::DBus)
        target_compile_definitions(test_qt_editor PRIVATE CEDAR_PORTAL_THEME)
    endif()
    add_test(NAME qt_editor_workflows COMMAND test_qt_editor)
    set_tests_properties(qt_editor_workflows PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(test_qt_frontend tests.cpp Appearance.cpp Appearance.h Simulation.cpp Simulation.h)
    target_link_libraries(test_qt_frontend PRIVATE CedarModel Qt6::Core Qt6::Gui)
    if(UNIX AND NOT APPLE)
        target_link_libraries(test_qt_frontend PRIVATE Qt6::DBus)
        target_compile_definitions(test_qt_frontend PRIVATE CEDAR_PORTAL_THEME)
    endif()
    target_compile_definitions(test_qt_frontend PRIVATE CEDAR_SOURCE_RES="${CEDAR_SOURCE}/res")
    add_test(NAME qt_frontend_invariants COMMAND test_qt_frontend)
    set_tests_properties(qt_frontend_invariants PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

# Keep crash reports useful in distributed optimized builds.
if(MSVC)
    target_compile_options(CedarLogic PRIVATE /Z7)
    target_compile_options(CedarModel PRIVATE /Z7)
    target_link_options(CedarLogic PRIVATE /DEBUG:FULL)
    target_link_libraries(CedarLogic PRIVATE dbghelp)
else()
    target_compile_options(CedarLogic PRIVATE -g)
    target_compile_options(CedarModel PRIVATE -g)
    if(UNIX AND NOT APPLE)
        target_link_options(CedarLogic PRIVATE -rdynamic)
    endif()
endif()
